Mathias Fleury (Nov 15 2026 at 16:40):

Here is the complete example:

lemma
  fixes X:: "'a set"
  fixes Y:: "'b set"
  fixes f:: "'a ⇒ 'b"
  fixes g:: "'b ⇒ 'a"
  assumes injf: "inj_on f X"
  assumes injg: "inj_on g Y"
  shows False
proof -
  define R where Rdef: "R = X - g`Y"
  define h where hdef: "h = g ∘ f"
  define Xset where ‹Xset = rec_nat (X - g`Y) (λ_ a. h ` a)›
  have [simp]: ‹Xset 0 = X - g`Y› and ‹Xset (Suc n)  = h`(Xset n)› for n
    unfolding Xset_def by auto
